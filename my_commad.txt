./.venv/Scripts/activate

python -m pip install .\tools\
python -m pip install -e .\design\

// Create Global Secrets
python -m ectf25_design.gen_secrets -f global.secrets 1 3 4

// Create Docker Image
docker build -t build-decoder .\decoder

// Build Docker
docker run --rm -v .\decoder/:/decoder -v .\global.secrets:/global.secrets -v .\deadbeef_build:/out -e DECODER_ID=0xdeadbeef build-decoder

// Create Subscription
python -m ectf25_design.gen_subscription global.secrets -f deadbeef_c1.sub 0xDEADBEEF 32 128 1

// Flashing
python -m ectf25.utils.flash .\deadbeef_build\max78000.bin COM6

// Testing Subscription update
python -m ectf25.tv.subscribe .\deadbeef_c1.sub COM6 
python -m ectf25.tv.list COM6

// Testing Decoder
python -m ectf25.utils.tester --port COM6 -s .\global.secrets rand -c 1 -f 64
